function loadSavedLogin() {
  const su = localStorage.getItem('rememberedUser');
  if (localStorage.getItem('rememberMe') === 'true' && su) {
    document.getElementById('loginUser').value = su;
    document.getElementById('rememberCheckbox').checked = true;
  }
}

function doLogin() {
  const u = document.getElementById('loginUser').value.trim();
  const p = document.getElementById('loginPass').value.trim();
  const remember = document.getElementById('rememberCheckbox').checked;
  const err = document.getElementById('loginError');

  if (USERS[u] && USERS[u].pass === p) {
    currentUser = USERS[u];
    err.style.display = 'none';
    if (remember) {
      localStorage.setItem('rememberedUser', u);
      localStorage.setItem('rememberMe', 'true');
    } else {
      localStorage.removeItem('rememberedUser');
      localStorage.setItem('rememberMe', 'false');
    }
    document.getElementById('loginScreen').style.display = 'none';
    document.getElementById('mainApp').style.display = 'block';
    document.getElementById('userDisplay').textContent = currentUser.name;
    applyRole();
    if (currentUser.role === 'center') {
      const sc = document.getElementById('srchCenter');
      for (let o of sc.options) if (o.value === currentUser.name) { sc.value = o.value; break; }
    }
    showPage('search');
    if (currentUser.role === 'admin' || currentUser.role === 'office') { svInit(); lvInit(); }
    sessHeartbeat();
    if (window.sessHeartbeatTimer) clearInterval(window.sessHeartbeatTimer);
    window.sessHeartbeatTimer = setInterval(sessHeartbeat, 60000);
    sessLoadActiveCount();
    hInitSystem();
    showToast('✅ مرحباً ' + currentUser.name, 'success');
  } else {
    err.style.display = 'block';
    setTimeout(() => err.style.display = 'none', 3000);
  }
}

function doLogout() {
  if (window.sessHeartbeatTimer) clearInterval(window.sessHeartbeatTimer);
  currentUser = null;
  document.getElementById('loginScreen').style.display = 'flex';
  document.getElementById('mainApp').style.display = 'none';
  document.getElementById('loginUser').value = '';
  document.getElementById('loginPass').value = '';
  showToast('👋 تم تسجيل الخروج', 'info');
}

function applyRole() {
  const isAdmin  = currentUser && currentUser.role === 'admin';
  const isOffice = currentUser && currentUser.role === 'office';
  const isCenter = currentUser && currentUser.role === 'center';

  // admin-only: فقط للمشرفين (إدارة الحافز + الجلسات)
  document.querySelectorAll('.admin-only').forEach(el => {
    el.style.display = isAdmin ? '' : 'none';
  });

  // admin-office-only: للمشرفين والمكتب (باقي أقسام الإدارة)
  document.querySelectorAll('.admin-office-only').forEach(el => {
    el.style.display = (isAdmin || isOffice) ? '' : 'none';
  });

  // center-hafez-only: فقط لمستخدمي المراكز
  document.querySelectorAll('.center-hafez-only').forEach(el => {
    el.style.display = isCenter ? '' : 'none';
  });

  // إظهار جرس الإشعارات فقط للمشرف
  const bell = document.getElementById('hNotifBell');
  if (bell) bell.style.display = isAdmin ? 'flex' : 'none';

  mobApplyRole();
}

  if (name === 'hafez-employees') hLoadAllEmps();
  if (name === 'hafez-center')    hLoadCenterData();
}

function escapeHtml(text) {
  if (!text) return '';
  return String(text).replace(/[&<>]/g, m => ({ '&':'&amp;', '<':'&lt;', '>':'&gt;' }[m]));
}

async function doSearch() {
  if (isSearching) { showToast('⚠️ انتظر البحث السابق', 'error'); return; }
  const center = document.getElementById('srchCenter').value, type = document.getElementById('srchType').value, val = document.getElementById('srchInput').value.trim();
  if (!center) { showToast('⚠️ اختر المركز أولاً', 'error'); return; }
  if (!val) { showToast('⚠️ أدخل اسم أو رقم طلب', 'error'); return; }
  addToHistory(val);
  const api = type === 'survey' ? SURVEY_API : LEVELING_API;
  isSearching = true; const btn = document.getElementById('srchBtn'); btn.textContent = '⏳ جاري البحث...'; btn.disabled = true;
  document.getElementById('srchResults').innerHTML = '<div class="no-results">⏳ جاري البحث...</div>'; document.getElementById('srchCount').innerHTML = '';
  try {
    const res = await fetch(`${api}?center=${encodeURIComponent(center)}&search=${encodeURIComponent(val)}&limit=10&t=${Date.now()}`);
    const data = await res.json();
    if (!data || !data.length) { document.getElementById('srchResults').innerHTML = '<div class="no-results">🔍 لا توجد نتائج</div>'; document.getElementById('srchCount').innerHTML = '❌ لا نتائج'; showToast('❌ لا توجد نتائج', 'error'); }
    else {
      document.getElementById('srchCount').innerHTML = `✅ ${data.length} نتيجة`;
      let html = '';
      for (const row of data) {
        const isPaid = !row[12] || row[12] === '' || row[12] === 'مستوفى'; const cls = isPaid ? 'paid' : 'unpaid';
        const addr = (row[8] && String(row[8]).trim()) ? row[8] : (row[1] || '-');
        html += `<div class="result-card ${cls}"><div class="card-header"><span class="request-number">📄 ${esc(row[2])}</span><span class="center-badge">🏢 ${esc(row[1])}</span></div><div class="card-body"><div class="info-row"><div class="info-icon">👤</div><div class="info-content"><div class="info-label">اسم الطالب</div><div class="info-value">${esc(row[3])}</div></div></div><div class="info-row"><div class="info-icon">📍</div><div class="info-content"><div class="info-label">العنوان</div><div class="info-value">${esc(addr)}</div></div></div><div class="info-row"><div class="info-icon">📅</div><div class="info-content"><div class="info-label">تاريخ الإصدار</div><div class="info-value">${fmtDate(row[5])}</div></div></div><div class="receiver-box"><div class="receiver-name">✍️ ${esc(row[6] || 'مدير الإدارة')}</div><div class="receiver-date">📅 ${fmtDate(row[7])}</div></div></div></div>`;
      }
      document.getElementById('srchResults').innerHTML = html; showToast(`✅ ${data.length} نتيجة`, 'success');
    }
  } catch (e) { document.getElementById('srchResults').innerHTML = '<div class="no-results">❌ خطأ في الاتصال</div>'; showToast('❌ خطأ في الاتصال', 'error'); }
  finally { isSearching = false; btn.textContent = '🔍 بحث'; btn.disabled = false; }
}

function hDedupRows(rows){
  const seenId=new Set(),seenName=new Set(),out=[];
  (rows||[]).forEach(r=>{
    if(!r)return;
    const id=String(r.nationalId||'').trim();
    const nm=hNormAr(r.name||'');
    if(id&&seenId.has(id))return;
    if(!id&&nm&&seenName.has(nm))return;
    if(id)seenId.add(id);if(nm)seenName.add(nm);
    out.push(r);
  });
  return out;
}

// التعديل 3: طباعة مجمعة — يظهر السنوات بها كشوف معتمدة فقط
// 
function getAvailableYearsList() {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheets = ss.getSheets();
    const yearsMap = {};
    
    for (const sheet of sheets) {
      const sheetName = sheet.getName();
      
      // محاولة استخراج السنة من اسم الورقة بأي صيغة
      let year = null;
      let month = null;
      
      // محاولة إيجاد رقم سنة (2024, 2025, 2026...)
      const yearMatch = sheetName.match(/(20\d{2})/);
      if (yearMatch) {
        year = yearMatch[1];
      }
      
      // محاولة إيجاد شهر (01-12)
      const monthMatch = sheetName.match(/(?:-|_|\s)(\d{1,2})(?:-|_|\s|$)/);
      if (monthMatch) {
        month = monthMatch[1].padStart(2, '0');
      }
      
      if (!year) continue; // تخطي الأوراق التي لا تحتوي على سنة
      
      // قراءة البيانات من الورقة
      const data = sheet.getDataRange().getValues();
      if (data.length < 2) continue;
      
      // البحث عن عمود الحالة (حاول إيجاده من الصف الأول)
      let statusColIndex = -1;
      const headers = data[0] || [];
      for (let i = 0; i < headers.length; i++) {
        const header = String(headers[i] || '').toLowerCase();
        if (header.includes('حالة') || header.includes('status')) {
          statusColIndex = i;
          break;
        }
      }
      
      // إذا لم نجد عمود حالة، نتحقق من وجود أي بيانات (نعتبر الكشف معتمد إذا كان يحتوي على بيانات)
      let hasApproved = false;
      
      if (statusColIndex !== -1) {
        // التحقق من وجود كشوف معتمدة
        hasApproved = data.slice(1).some(row => {
          const status = String(row[statusColIndex] || '').toLowerCase();
          return status === 'معتمد' || status === 'إرسال نهائي' || status === 'مغلق' || 
                 status === 'approved' || status === 'تم الاعتماد';
        });
      } else {
        // إذا لم نجد عمود حالة، نعتبر الكشف معتمد إذا كان يحتوي على بيانات
        hasApproved = data.slice(1).length > 0;
      }
      
      if (hasApproved) {
        if (!yearsMap[year]) {
          yearsMap[year] = { year: year, months: new Set(), sheetCount: 0 };
        }
        if (month) {
          yearsMap[year].months.add(`${year}-${month}`);
        }
        yearsMap[year].sheetCount++;
      }
    }
    
    const years = Object.values(yearsMap).map(y => ({
      year: y.year,
      monthCount: y.months.size,
      sheetCount: y.sheetCount
    })).sort((a, b) => b.year - a.year);
    
    // إذا لم يتم العثور على أي سنة، نضيف سنوات افتراضية للتجربة
    if (years.length === 0) {
      // إضافة سنة 2025 و 2024 كتجربة (يمكن تعديلها حسب الحاجة)
      years.push({ year: '2025', monthCount: 1, sheetCount: 1 });
      years.push({ year: '2024', monthCount: 1, sheetCount: 1 });
    }
    
    return ContentService.createTextOutput(JSON.stringify({
      success: true,
      years: years,
      debug: { totalSheets: sheets.length, processedSheets: Object.keys(yearsMap).length }
    })).setMimeType(ContentService.MimeType.JSON);
    
  } catch(e) {
    return ContentService.createTextOutput(JSON.stringify({
      success: false,
      error: e.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }

function getApprovedSheetsByYear(params) {
  try {
    const year = params.year;
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheets = ss.getSheets();
    const resultSheets = [];
    
    for (const sheet of sheets) {
      const sheetName = sheet.getName();
      
      // التحقق من أن الورقة تحتوي على السنة المطلوبة
      if (!sheetName.includes(year)) continue;
      
      // استخراج المعلومات من اسم الورقة
      let month = '';
      let center = '';
      let type = '';
      
      // محاولة استخراج الشهر
      const monthMatch = sheetName.match(/(20\d{2})[-_](\d{1,2})/);
      if (monthMatch) {
        month = `${monthMatch[1]}-${monthMatch[2].padStart(2, '0')}`;
      } else {
        const anyMonth = sheetName.match(/(\d{1,2})(?:-|_|\s|$)/);
        if (anyMonth) {
          month = `${year}-${anyMonth[1].padStart(2, '0')}`;
        } else {
          month = year;
        }
      }
      
      // محاولة استخراج المركز
      const centersList = ['العدوة', 'مغاغة', 'بني مزار', 'مطاي', 'سمالوط', 'المنيا', 'ابوقرقاص', 'ملوي', 'ديرمواس', 'أبو قرقاص'];
      for (const c of centersList) {
        if (sheetName.includes(c)) {
          center = c;
          break;
        }
      }
      
      // محاولة استخراج النوع
      if (sheetName.includes('بنية') || sheetName.includes('معلوماتية')) {
        type = 'بنية معلوماتية';
      } else if (sheetName.includes('أجهزة') || sheetName.includes('معاونة')) {
        type = 'أجهزة معاونة';
      } else {
        type = 'كشف';
      }
      
      // قراءة البيانات
      const data = sheet.getDataRange().getValues();
      if (data.length < 2) continue;
      
      // البحث عن عمود الحالة
      let statusColIndex = -1;
      const headers = data[0] || [];
      for (let i = 0; i < headers.length; i++) {
        const header = String(headers[i] || '').toLowerCase();
        if (header.includes('حالة') || header.includes('status')) {
          statusColIndex = i;
          break;
        }
      }
      
      // حساب عدد الموظفين الموزع عليهم (أي صف به مبلغ > 0)
      let employeeCount = 0;
      let amountColIndex = -1;
      
      // البحث عن عمود المبلغ
      for (let i = 0; i < headers.length; i++) {
        const header = String(headers[i] || '').toLowerCase();
        if (header.includes('مبلغ') || header.includes('المبلغ') || header.includes('amount')) {
          amountColIndex = i;
          break;
        }
      }
      
      for (let i = 1; i < data.length; i++) {
        const row = data[i];
        
        // التحقق من الحالة إذا وجدنا عمود الحالة
        let isApproved = true;
        if (statusColIndex !== -1) {
          const status = String(row[statusColIndex] || '').toLowerCase();
          isApproved = status === 'معتمد' || status === 'إرسال نهائي' || status === 'مغلق' || 
                       status === 'approved' || status === 'تم الاعتماد';
        }
        
        // التحقق من وجود مبلغ إذا وجدنا عمود المبلغ
        let hasAmount = true;
        if (amountColIndex !== -1) {
          const amount = parseFloat(row[amountColIndex]);
          hasAmount = !isNaN(amount) && amount > 0;
        }
        
        if (isApproved && hasAmount) {
          employeeCount++;
        }
      }
      
      if (employeeCount > 0) {
        resultSheets.push({
          sheetId: sheet.getSheetId().toString(),
          sheetName: sheetName,
          month: month,
          center: center || 'غير محدد',
          type: type,
          employeeCount: employeeCount
        });
      }
    }
    
    return ContentService.createTextOutput(JSON.stringify({
      success: true,
      sheets: resultSheets,
      debug: { year: year, totalSheets: sheets.length, foundSheets: resultSheets.length }
    })).setMimeType(ContentService.MimeType.JSON);
    
  } catch(e) {
    return ContentService.createTextOutput(JSON.stringify({
      success: false,
      error: e.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}

function pmUpdateMonths() {
  const year = document.getElementById('pmYear')?.value;
  if (!year) return;
  const sel = document.getElementById('pmMonth');
  if (!sel) return;
  const today = new Date();
  const maxMo = year == today.getFullYear() ? today.getMonth() + 1 : 12;
  let html = '<option value="">-- كل الشهور --</option>';
  for (let mo = 1; mo <= maxMo; mo++) {
    const moStr = String(mo).padStart(2, '0');
    html += `<option value="${moStr}">${hFmtMonth(`${year}-${moStr}`)}</option>`;
  }
  sel.innerHTML = html;
}

function _hFileToBase64(file){
  return new Promise((resolve,reject)=>{
    const r = new FileReader();
    r.onload = ()=>{
      const s = String(r.result||'');
      const b64 = s.includes(',') ? s.split(',')[1] : s;
      resolve(b64);
    };
    r.onerror = ()=>reject(r.error);
    r.readAsDataURL(file);
  });

async function hHandleAttachFile(){
  const inp = document.getElementById('hAttachFileInput');
  const files = Array.from(inp.files||[]);
  if(!files.length) return;
  const status = document.getElementById('hAttachUploadStatus');
  let ok=0, fail=0;
  for(const f of files){
    if(f.size > 10*1024*1024){ fail++; status.innerHTML = `<span style="color:#dc2626">❌ ${escapeHtml(f.name)} أكبر من 10 ميجا</span>`; continue; }
    status.innerHTML = `<span style="color:#1e3a5f">⏳ جارٍ رفع: ${escapeHtml(f.name)}...</span>`;
    try{
      const b64 = await _hFileToBase64(f);
      const r = await hApiCall('uploadAttachment',{
        sheetId: hAttCurSid, fileName: f.name, mimeType: f.type, base64: b64,
        user: (currentUser&&currentUser.name)||'system'
      });
      if(r.success) ok++; else { fail++; status.innerHTML = `<span style="color:#dc2626">❌ ${escapeHtml(r.message||'فشل الرفع')}</span>`; }
    }catch(e){ fail++; }
  }
  inp.value='';
  status.innerHTML = `<span style="color:${fail?'#d97706':'#16a085'}">تم رفع ${ok} ملف${fail?` — فشل ${fail}`:''}</span>`;
  await hRefreshAttachList();
}

async function hDeleteAttachment(fileId, rowIndex, name){
  if(!confirm(`حذف المرفق "${name}"؟`)) return;
  const r = await hApiCall('deleteAttachment',{fileId, rowIndex, user:(currentUser&&currentUser.name)||'system'});
  if(r.success){ hToast('🗑️ تم حذف المرفق','success'); await hRefreshAttachList(); }
  else hToast(r.message||'خطأ فى الحذف','error');
}

  tmp.onchange = async ()=>{
    const f = tmp.files && tmp.files[0]; if(!f) return;
    if(f.size > 10*1024*1024){ hToast('الملف أكبر من 10 ميجا','error'); return; }
    const status = document.getElementById('hAttachUploadStatus');
    status.innerHTML = `<span style="color:#1e3a5f">⏳ جارٍ الاستبدال...</span>`;
    try{
      const b64 = await _hFileToBase64(f);
      const r = await hApiCall('replaceAttachment',{
        sheetId: hAttCurSid, fileId, rowIndex,
        fileName: f.name, mimeType: f.type, base64: b64,
        user: (currentUser&&currentUser.name)||'system'
      });
      if(r.success){ status.innerHTML = `<span style="color:#16a085">✅ تم الاستبدال</span>`; await hRefreshAttachList(); }
      else { status.innerHTML = `<span style="color:#dc2626">❌ ${escapeHtml(r.message||'فشل')}</span>`; }
    }catch(e){ hToast('خطأ: '+e.message,'error'); }
  };
  tmp.click();
}
</script>
